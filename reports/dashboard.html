import streamlit as st import pandas as pd import json from datetime import datetime # --- CARREGAR DADOS DO RELATÓRIO JSON --- try: with open("reports/security_report.json", "r", encoding="utf-8") as f: SECURITY_DATA = json.load(f) except FileNotFoundError: # Fallback caso o arquivo ainda não tenha sido gerado SECURITY_DATA = { "timestamp": str(datetime.now()), "region": "us-east-1", "total_findings": 0, "security_hub_summary": {"critical": 0, "high": 0, "medium": 0, "low": 0}, "findings": [] } # --- CÁLCULO DE SECURITY SCORE DINÂMICO --- def calculate_dynamic_score(summary): score = 100 score -= summary.get("critical", 0) * 25 score -= summary.get("high", 0) * 15 score -= summary.get("medium", 0) * 5 return max(score, 0) current_score = calculate_dynamic_score(SECURITY_DATA["security_hub_summary"]) # --- CONFIGURAÇÃO DA PÁGINA --- st.set_page_config( page_title="AWS Cyber Defense Dashboard", page_icon="📊", layout="wide" ) # --- ESTILO VISUAL CUSTOMIZADO (CSS) --- st.markdown(""" """, unsafe_allow_html=True) # --- MENU LATERAL DE NAVEGAÇÃO --- st.sidebar.title("🛡️ CSOC Enterprise") st.sidebar.markdown(f"**Região:** `{SECURITY_DATA['region']}`") st.sidebar.markdown(f"**Último Scan:** `{SECURITY_DATA['timestamp']}`") st.sidebar.markdown("---") page = st.sidebar.radio("Seções do Dashboard", [ "📊 Dashboard", "🚨 Security Hub Findings", "🎯 MITRE ATT&CK", "🧠 Attack Path", "📄 JSON Payload View" ]) st.sidebar.markdown("---") st.sidebar.info("🏢 **Engine:** Pipeline JSON + Streamlit\n🔒 **Status:** Sincronizado") # ========================================== # 1.
